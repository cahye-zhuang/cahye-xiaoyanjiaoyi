    static getDefaultData() {
      return {
        version: DATA_VERSION, initialCapital: 5, orders: [],
        realmSettings: null, liquidationEvents: [],
        maxRealmAchieved: "斗之气一段", journals: [],

    static load() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) {
          let data = JSON.parse(raw);
          if (data.version !== DATA_VERSION) data = DataManager.migrate(data);
          return data;
        }
      } catch(e) {}
      return DataManager.getDefaultData();
    }
    static migrate(old) {
      const def = DataManager.getDefaultData();
      Object.keys(def).forEach(k => { if (!(k in old)) old[k] = def[k]; });
      old.version = DATA_VERSION;
      (old.orders || []).forEach(o => {
        o.profit = Number(o.profit) || 0;
        o.fee = Number(o.fee) || 0;
        if (!o.tag) o.tag = '合约';
      });
      return old;
    }
    static save(data) {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
      } catch(e) {
        if (e.name === 'QuotaExceededError') {
          showToast('存储空间不足，请清理旧日志或导出数据');
        }
      }
    }
    static backup() {
      const blob = new Blob([JSON.stringify(appData)], {type:'application/json'});
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = 'trading_backup.json';
      a.click();
    }
    static importFile(file, onSuccess) {
      const reader = new FileReader();
      reader.onload = (ev) => {
        try {
          const data = JSON.parse(ev.target.result);
          if (typeof data.initialCapital !== 'number' || !Array.isArray(data.orders)) throw new Error();
          return onSuccess(DataManager.migrate(data));
        } catch(e) {
          showToast('文件格式无效');
        }
      };
      reader.readAsText(file);
    }
  }

  // ==================== 统计引擎 ====================
  class StatsEngine {
    static getOrdersInPeriod(orders, period) {
      let filtered = [...orders];
      const now = new Date();
      if (period === 'today') {
        const start = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
        filtered = filtered.filter(o => new Date(o.dateTime).getTime() >= start);
      } else if (period === 'week') {
        const day = now.getDay() || 7;
        const start = new Date(now.getFullYear(), now.getMonth(), now.getDate() - day + 1).getTime();
        filtered = filtered.filter(o => new Date(o.dateTime).getTime() >= start);
      } else if (period === 'month') {
        const start = new Date(now.getFullYear(), now.getMonth(), 1).getTime();
        filtered = filtered.filter(o => new Date(o.dateTime).getTime() >= start);
      }
      return filtered.sort((a,b) => new Date(a.dateTime) - new Date(b.dateTime));
    }
    static compute(orders, initialCapital, realms) {
      const allProfit = orders.reduce((s,o) => s + o.profit, 0);
      const capital = initialCapital + allProfit;
      const win = orders.filter(o => o.profit > 0);
      const loss = orders.filter(o => o.profit < 0);
      const winCount = win.length, lossCount = loss.length;
      const total = winCount + lossCount;
      const winRate = total ? (winCount / total * 100) : 0;
      const avgWin = winCount ? win.reduce((s,o)=>s+o.profit,0)/winCount : 0;
      const avgLoss = lossCount ? Math.abs(loss.reduce((s,o)=>s+o.profit,0)/lossCount) : 0;
      const profitRatio = (avgLoss===0 || avgWin===0) ? null : avgWin/avgLoss;

      let maxWinStreak=0, maxLossStreak=0, curWin=0, curLoss=0;
      orders.forEach(o => {
        if(o.profit>0){ curWin++; curLoss=0; maxWinStreak=Math.max(maxWinStreak,curWin); }
        else if(o.profit<0){ curLoss++; curWin=0; maxLossStreak=Math.max(maxLossStreak,curLoss); }
        else{ curWin=0; curLoss=0; }
      });
      const avgLeverage = orders.length ? orders.reduce((s,o)=>s+(o.leverage||1),0)/orders.length : 1;
      const totalFee = orders.reduce((s,o)=> s + (o.fee||0), 0);

      // 最大回撤
      let peak = initialCapital, mdd = 0;
      let runningCapital = initialCapital;
      const sortedForDD = [...orders].sort((a,b) => new Date(a.dateTime) - new Date(b.dateTime));
      sortedForDD.forEach(o => {
        runningCapital += o.profit;
        if (runningCapital > peak) peak = runningCapital;
        const dd = (peak - runningCapital) / peak * 100;
        if (dd > mdd) mdd = dd;
      });

      // 境界
      let realmObj = realms[0];
      for(let i=realms.length-1; i>=0; i--) if(capital >= realms[i].min) { realmObj = realms[i]; break; }
      const nextIdx = realms.indexOf(realmObj)+1;
      const nextRealm = nextIdx < realms.length ? realms[nextIdx] : null;
      const progress = nextRealm ? Math.min(100, ((capital-realmObj.min)/(nextRealm.min-realmObj.min))*100) : 100;

      // 风险
      const capRatio = capital / initialCapital;
      let riskLevel='safe', riskText='安全', riskDetail='本金充足';
      if(capRatio<0.2){ riskLevel='danger'; riskText='高危'; riskDetail='接近爆仓'; }
      else if(capRatio<0.5){ riskLevel='warning'; riskText='警告'; riskDetail='回撤超50%'; }
      else if(capRatio<0.8){ riskLevel='warning'; riskText='注意'; riskDetail='回撤超20%'; }

      // 币种统计
      const symMap = {};
      orders.forEach(o => {
        const s = o.symbol;
        if(!symMap[s]) symMap[s] = { total:0, wins:0, profit:0 };
        symMap[s].total++;
        if(o.profit>0) symMap[s].wins++;
        symMap[s].profit += o.profit;
      });
      const symbolWin = Object.entries(symMap).map(([s,d])=>({
        symbol:s, ...d, winRate: d.total?(d.wins/d.total*100):0
      })).sort((a,b)=>b.winRate-a.winRate);
      const symbolProfit = [...symbolWin].sort((a,b)=>b.profit-a.profit);

      return {
        capital, total, winCount, lossCount, winRate,
        profitRatio, maxWinStreak, maxLossStreak, avgLeverage, totalFee,
        realm: realmObj.name, nextRealm: nextRealm?.name||'—', progress,
        riskLevel, riskText, riskDetail, symbolWin, symbolProfit,
        mdd
      };
    }
  }

  function calcFee(cap, lev, gross, openT, closeT) {
    const openVal = cap * lev;
    const closeVal = openVal;
    const openRate = openT === 'maker' ? 0.0002 : 0.0005;
    const closeRate = closeT === 'maker' ? 0.0002 : 0.0005;
    const fee = openVal * openRate + closeVal * closeRate;
    return { fee, netProfit: gross - fee };
  }

  function drawEquityChart(stats) {
    const canvas = document.getElementById('equityChart');
    if(!canvas) return;
    const ctx = canvas.getContext('2d');
    ctx.clearRect(0,0,canvas.width,canvas.height);
    const orders = StatsEngine.getOrdersInPeriod(appData.orders, currentPeriod);
    if(orders.length < 2) {
      ctx.fillStyle = getComputedStyle(document.body).getPropertyValue('--text-secondary');
      ctx.fillText('数据不足',20,50);
      return;
    }
    let points = [], cum = appData.initialCapital;
    const sortedOrders = orders.sort((a,b) => new Date(a.dateTime)-new Date(b.dateTime));
    sortedOrders.forEach(o => { cum += o.profit; points.push(cum); });
    const max = Math.max(...points, appData.initialCapital);
    const min = Math.min(...points, appData.initialCapital);
    const range = max-min || 1;
    ctx.strokeStyle = getComputedStyle(document.body).getPropertyValue('--accent');
    ctx.lineWidth = 2;
    ctx.beginPath();
    points.forEach((p,i) => {
      const x = (i/(points.length-1))*(canvas.width-20)+10;
      const y = canvas.height - ((p-min)/range)*(canvas.height-20)-10;
      i===0 ? ctx.moveTo(x,y) : ctx.lineTo(x,y);
    });
    ctx.stroke();
    ctx.strokeStyle = getComputedStyle(document.body).getPropertyValue('--border-color');
    ctx.lineWidth = 0.5;
    for(let i=0; i<=4; i++) {
      let y = 10 + i*(canvas.height-20)/4;
      ctx.beginPath(); ctx.moveTo(10,y); ctx.lineTo(canvas.width-10,y); ctx.stroke();
    }
  }

  function getTradeAdvice(stats) {
    if(stats.total < 5) return "数据不足，积累≥5笔。保持轻仓，杠杆≤5倍。";
    if(appData.lockAfterLosses && stats.maxLossStreak >= appData.maxLossStreakLock) return "🔒 连亏达上限，请休息复盘。";
    let adv = "";
    if(stats.profitRatio && stats.profitRatio>0) {
      const kelly = Math.min(0.25, Math.max(0, stats.winRate/100 - (1-stats.winRate/100)/stats.profitRatio));
      adv += `📊 凯利仓位：${(kelly*100).toFixed(1)}% 内。`;
    } else adv += "⚠️ 盈亏比不佳，减仓。";
    if(stats.maxLossStreak>=3) adv += " 🔴 连亏3次，休息。";
    if(stats.avgLeverage>10) adv += " ⚡ 杠杆偏高。";
    if(stats.riskLevel==='danger') adv += " 🆘 爆仓边缘！";
    return adv || "✅ 状态良好。";
  }

  function submitTrade() {
    const sym = document.getElementById('tradeSymbolInput').value.trim().toUpperCase();
    if(!sym) return alert('请选择币种');
    if(!appData.customSymbols.includes(sym)) { appData.customSymbols.push(sym); DataManager.save(appData); }
    const positionSize = parseFloat(document.getElementById('tradePosition').value);
    if(isNaN(positionSize) || positionSize < 0) return alert('本金不能为负数');
    const leverage = parseInt(document.getElementById('tradeLeverage').value);
    if(leverage < 1 || leverage > 100) return alert('杠杆范围为1-100');
    const gross = parseFloat(document.getElementById('tradeGrossProfit').value);
    if(isNaN(gross)) return alert('请填写毛盈亏');
    const box = document.getElementById('netProfitDisplay');
    if(!box.dataset.netProfit) updateFeePreview();
    const netProfit = parseFloat(box.dataset.netProfit);
    const fee = parseFloat(box.dataset.fee);
    const order = {
      id: editingOrderId || Date.now().toString(36)+Math.random().toString(36).substr(2,5),
      dateTime: new Date().toISOString(),
      symbol: sym, direction: document.getElementById('tradeDirection').value,
      positionSize, leverage, profit: netProfit, fee,
      openFee: document.getElementById('openFeeType').value,
      closeFee: document.getElementById('closeFeeType').value,
      entry: document.getElementById('entryPrice').value,
      exit: document.getElementById('exitPrice').value,
      stopLoss: document.getElementById('stopLoss').value,
      takeProfit: document.getElementById('takeProfit').value,
      tag: document.getElementById('tradeTag').value,
      note: document.getElementById('tradeNote').value
    };
    if(editingOrderId) {
      const idx = appData.orders.findIndex(o => o.id === editingOrderId);
      if(idx !== -1) appData.orders[idx] = order;
    } else {
      appData.orders.push(order);
    }
    DataManager.save(appData);
    clearTradeForm();
    editingOrderId = null;
    document.getElementById('submitTradeBtn').textContent = '📈 记录交易';
    refreshUI();
    showToast('✅ 已保存');
  }

  function clearTradeForm() {
    ['tradeSymbolInput','tradePosition','tradeGrossProfit','entryPrice','exitPrice','stopLoss','takeProfit','tradeNote'].forEach(id => document.getElementById(id).value='');
    document.getElementById('tradeLeverage').value='1';
    document.querySelectorAll('.lev-btn').forEach(b => b.classList.remove('selected'));
    document.querySelector('.lev-btn[data-lev="1"]').classList.add('selected');
    document.getElementById('netProfitDisplay').style.display='none';
    document.getElementById('detailsBody').classList.remove('open');
    document.getElementById('toggleArrow').textContent = '▶';
  }

  function editOrder(id) {
    const order = appData.orders.find(o => o.id === id);
    if(!order) return;
    editingOrderId = id;
    document.getElementById('tradeSymbolInput').value = order.symbol;
    document.getElementById('tradeDirection').value = order.direction;
    document.getElementById('tradePosition').value = order.positionSize;
    document.getElementById('tradeGrossProfit').value = (order.profit + (order.fee||0)).toFixed(4);
    document.getElementById('tradeLeverage').value = order.leverage;
    document.querySelectorAll('.lev-btn').forEach(b => b.classList.remove('selected'));
    const levBtn = document.querySelector(`.lev-btn[data-lev="${order.leverage}"]`);
    if(levBtn) levBtn.classList.add('selected');
    document.getElementById('openFeeType').value = order.openFee || 'taker';
    document.getElementById('closeFeeType').value = order.closeFee || 'taker';
    document.getElementById('entryPrice').value = order.entry || '';
    document.getElementById('exitPrice').value = order.exit || '';
    document.getElementById('stopLoss').value = order.stopLoss || '';
    document.getElementById('takeProfit').value = order.takeProfit || '';
    document.getElementById('tradeTag').value = order.tag || '合约';
    document.getElementById('tradeNote').value = order.note || '';
    if(!document.getElementById('detailsBody').classList.contains('open')) {
      document.getElementById('detailsBody').classList.add('open');
      document.getElementById('toggleArrow').textContent = '▼';
    }
    updateFeePreview();
    document.getElementById('submitTradeBtn').textContent = '✏️ 更新交易';
    document.getElementById('page-trade').scrollIntoView({behavior:'smooth'});
  }

  // 日志
  function openJournalModal(idx=null) {
    journalEditIndex = idx;
    document.getElementById('journalModalTitle').textContent = idx!==null ? '编辑日志' : '新日志';
    document.getElementById('journalContent').value = idx!==null ? appData.journals[idx].content : '';
    document.getElementById('journalModal').style.display = 'flex';
  }
  function closeJournalModal() { document.getElementById('journalModal').style.display = 'none'; }
  function saveJournal() {
    const content = document.getElementById('journalContent').value.trim();
    if(!content) return;
    if(journalEditIndex !== null) appData.journals[journalEditIndex].content = content;
    else appData.journals.push({ dateTime: new Date().toISOString(), content });
    DataManager.save(appData);
    closeJournalModal();
    renderJournalList();
  }
  function handleJournalClick(e) {
    const idx = e.target.dataset.editj;
    if(idx !== undefined) openJournalModal(parseInt(idx));
    else if(e.target.dataset.delj !== undefined) {
      if(confirm('删除此日志？')) {
        appData.journals.splice(parseInt(e.target.dataset.delj),1);
        DataManager.save(appData);
        renderJournalList();
      }
    }
  }

  // 设置弹窗
  function openSettingsModal() {
    const modal = document.createElement('div'); modal.className='modal-overlay'; modal.style.display='flex';
    modal.innerHTML = `
      <div class="modal"><h3>设置</h3>
        <label>初始本金</label><input type="number" id="setInitial" value="${appData.initialCapital}">
        <div style="display:flex; align-items:center; justify-content:space-between; margin:12px 0;">
          <span>深色模式</span><div class="toggle-switch ${appData.theme==='dark'?'active':''}" id="setTheme"></div>
        </div>
        <div style="display:flex; align-items:center; justify-content:space-between; margin-bottom:12px;">
          <span>跟随系统</span><div class="toggle-switch ${appData.autoTheme?'active':''}" id="setAutoTheme"></div>
        </div>
        <label>连亏禁开单笔数</label><input type="number" id="setMaxLoss" value="${appData.maxLossStreakLock}">
        <div style="margin:16px 0; display:flex; gap:8px;">
          <button class="btn" id="resetBtn">重置挑战</button>
          <button class="btn" id="exportBtn">导出</button>
          <button class="btn" id="importBtn">导入</button><input type="file" id="importFile" style="display:none;" accept=".json">
        </div>
        <div style="display:flex; justify-content:flex-end; gap:8px;">
          <button class="btn" id="closeSetBtn">取消</button>
          <button class="btn btn-primary" id="saveSetBtn">保存</button>
        </div>
      </div>`;
    document.body.appendChild(modal);

    document.getElementById('setTheme').addEventListener('click', function(){ this.classList.toggle('active'); });
    document.getElementById('setAutoTheme').addEventListener('click', function(){ this.classList.toggle('active'); });
    document.getElementById('resetBtn').addEventListener('click', ()=>{
      if(confirm('确定重置所有数据？')) {
        appData = DataManager.getDefaultData();
        DataManager.save(appData);
        applyTheme();
        modal.remove();
        refreshUI();
      }
    });
    document.getElementById('closeSetBtn').addEventListener('click', ()=> modal.remove());
    document.getElementById('saveSetBtn').addEventListener('click', ()=>{
      appData.initialCapital = parseFloat(document.getElementById('setInitial').value)||5;
      appData.theme = document.getElementById('setTheme').classList.contains('active') ? 'dark' : 'light';
      appData.autoTheme = document.getElementById('setAutoTheme').classList.contains('active');
      appData.maxLossStreakLock = parseInt(document.getElementById('setMaxLoss').value)||3;
      if(appData.autoTheme) {
        appData.theme = window.matchMedia('(prefers-color-scheme: light)').matches ? 'light' : 'dark';
      }
      applyTheme();
      DataManager.save(appData);
      modal.remove();
      refreshUI();
    });
    document.getElementById('exportBtn').addEventListener('click', DataManager.backup);
    document.getElementById('importBtn').addEventListener('click', ()=> document.getElementById('importFile').click());
    document.getElementById('importFile').addEventListener('change', (e)=>{
      const file = e.target.files[0];
      if(!file) return;
      DataManager.importFile(file, (importedData) => {
        appData = importedData;
        DataManager.save(appData);
        applyTheme();
        modal.remove();
        refreshUI();
        showToast('导入成功');
      });
    });
  }

  // 最近订单的事件
  document.getElementById('recentOrders').addEventListener('click', (e) => {
    const orderId = e.target.closest('[data-order-id]')?.dataset.orderId;
    if(orderId && e.target.tagName !== 'BUTTON') editOrder(orderId);
    else if(e.target.dataset.delOrder) {
      if(confirm('删除此订单？')) {
        appData.orders = appData.orders.filter(o => o.id !== e.target.dataset.delOrder);
        DataManager.save(appData);
        refreshUI();
      }
    }
  });

  // Service Worker 注册
  if('serviceWorker' in navigator) {
    window.addEventListener('load', () => {
      // 简单的离线缓存
      navigator.serviceWorker.register(
        URL.createObjectURL(new Blob([`
          self.addEventListener('install', e => self.skipWaiting());
          self.addEventListener('activate', e => e.waitUntil(clients.claim()));
          self.addEventListener('fetch', e => e.respondWith(
            caches.match(e.request).then(r => r || fetch(e.request).then(resp => {
              if(resp.ok) {
                const clone = resp.clone();
                caches.open('v1').then(c => c.put(e.request, clone));
                return resp;
              }
              return resp;
            }).catch(() => caches.match(e.request)))
          ));
        `], {type:'application/javascript'}))
      ).catch(() => {});
    });
  }

  // 系统主题监听
  window.matchMedia('(prefers-color-scheme: light)').addEventListener('change', (e) => {
    if(appData.autoTheme) {
      appData.theme = e.matches ? 'light' : 'dark';
      applyTheme();
    }
  });

  bindEvents();
  refreshUI();
})();
